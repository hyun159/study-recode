```
파일 위치: /etc/nginx/nginx.conf

    ----------------------------------------------------------------------------

# 코어 설정
user nginx;                          # 워커 프로세스 실행 유저
worker_processes auto;               # auto = CPU 코어 수만큼 워커 생성
error_log /var/log/nginx/error.log;  # 에러 로그 경로
pid /run/nginx.pid;                  # PID 파일

    ----------------------------------------------------------------------------

# 커넥션 처리
# cpu core 당 tcp 연결 가능 횟수
# 
events {
    worker_connections 1024;         # 워커 1개당 최대 동시 커넥션
}

    ----------------------------------------------------------------------------

# 웹 설정 전체
http {
    # access_log에 기록될 log 형식
    log_format  main  '$remote_addr - $remote_user [$time_local] "$request" '
                      '$status $body_bytes_sent "$http_referer" '
                      '"$http_user_agent"';

    access_log  /var/log/nginx/access.log  main;

    ----------------------------------------------------------------------------

    # ssd -> 커널 메모리 -> 엔진엑스 워커 메모리 -> 커널 메모리 -> NIC
    # ssd -> 커널 메모리 -> NIC
    sendfile            on;          # 커널에서 직접 파일 전송 (빠름)

    ----------------------------------------------------------------------------

    # http 헤더와 바디를 웬만해서 패킷단위로 채워 보내기
    tcp_nopush          on;          # sendfile 쓸 때 패킷 묶어서 보냄
    tcp_nodelay         on;          # 작은 패킷 즉시 전송
    keepalive_timeout   65;          # 연결 유지 시간(초)

    ----------------------------------------------------------------------------

    # /etc/nginx/mime.types에 존재하는 MIME 타입 테이블 공간 할당
    # 더 다양한 데이터 형식을 처리할 수 있다.
    # 해시 테이블 최대 엔트리 4096byte
    types_hash_max_size 4096;


    # MIME 타입 테이블 경로
    include             /etc/nginx/mime.types;

    # 확장자 못찾으면 기본 타입은 설정
    # octet-stream: 알 수 없는 바이너리 데이터
    default_type        application/octet-stream;

    ----------------------------------------------------------------------------

    limit_req_zone $binary_remote_addr zone=mylimit:10m rate=10r/s;
               ─────────────────  ──────────────── ──────────
               클라이언트 IP 기준   메모리 10MB 할당  초당 10개 제한

    ----------------------------------------------------------------------------
    
    # 로드밸런싱
    # 동일한 서비스를 그룹화
    # api 서버들, 웹앱 서버들, Spring 서버들 ...
    # location의 proxy_pass에서 upstream 그룹 사용

    upstream api {

        server IP:Port;
        server IP:Port;
        server IP:Port;
    }

     upstream spring {

        server IP:Port;
        server IP:Port;
        server IP:Port;
    }

     # 이 시점 부터 conf.d 파일을 불러옴
    include /etc/nginx/conf.d/*.conf;
}

```

도메인 분기

```
파일 위치
/etc/nginx/conf.d/dns01.conf


server {
        listen       80;
        listen       [::]:80;
        server_name  DNS주소              # 모든 호스트 매칭 (기본값) : _;
        root         /usr/share/nginx/html;

        
        # 리버스 프록시
        location /01-url { proxy_pass http://api; }
        location /02-url { proxy_pass http://spring; }

        # 헬스 체크
        location /health { return 200 "ok"; }

        # 리다이렉트
        # 301: 리다이렉트 코드
        # $request_uri: nginx 내부 변수, 요청받은 url 임시 저장
        return 301 https://URL$request_uri;
    }
```
